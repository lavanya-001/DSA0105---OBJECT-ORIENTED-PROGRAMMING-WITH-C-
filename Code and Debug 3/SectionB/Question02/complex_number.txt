QUESTION 2: COMPLEX NUMBER SYSTEM

CLASS Complex

PRIVATE:
    REAL real
    REAL imag

PUBLIC:

    CONSTRUCTOR()
        real = 0.0
        imag = 0.0
    END CONSTRUCTOR

    CONSTRUCTOR(r AS REAL)
        real = r
        imag = 0.0
    END CONSTRUCTOR

    CONSTRUCTOR(r AS REAL, i AS REAL)
        real = r
        imag = i
    END CONSTRUCTOR

    OPERATOR +(REF other AS Complex) RETURNS Complex
        RETURN Complex(real + other.real,
                       imag + other.imag)
    END OPERATOR

    OPERATOR *(REF other AS Complex) RETURNS Complex
        REAL r
        REAL i

        r = (real * other.real) - (imag * other.imag)
        i = (real * other.imag) + (imag * other.real)

        RETURN Complex(r, i)
    END OPERATOR

    OPERATOR ==(REF other AS Complex) RETURNS BOOLEAN
        RETURN (real == other.real AND imag == other.imag)
    END OPERATOR

    METHOD Display()
        DISPLAY real, " + ", imag, "i"
    END METHOD

END CLASS


MAIN

    CREATE C1 AS Complex(3.0, 2.0)
    CREATE C2 AS Complex(1.0, 4.0)

    CREATE sum AS C1 + C2
    CREATE product AS C1 * C2

    DISPLAY "Addition:"
    sum.Display()

    DISPLAY "Multiplication:"
    product.Display()

    IF C1 == C2 THEN
        DISPLAY "Complex numbers are equal"
    ELSE
        DISPLAY "Complex numbers are not equal"
    ENDIF

END MAIN


DRY RUN TABLE

C1 = 3 + 2i
C2 = 1 + 4i

Multiplication:

          = 3 - 8
          = -5

Imaginary part = (3 * 4) + (2 * 1)

Addition:
